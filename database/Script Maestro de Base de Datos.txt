-- ==============================================================================
-- SCRIPT MAESTRO DE LIMPIEZA Y CREACIÓN DESDE CERO PARA DEN DEN MUSHI WEB
-- ==============================================================================

-- 0. Limpieza previa para evitar conflictos de ejecuciones anteriores
drop table if exists public.ice_candidates cascade;
drop table if exists public.friends cascade;
drop table if exists public.rooms cascade;
drop table if exists public.profiles cascade;

-- ==============================================================================
-- TABLA 1: PERFILES DE USUARIO (public.profiles)
-- Enlaza directamente con el sistema de autenticación seguro de Supabase (auth.users)
-- ==============================================================================
create table public.profiles (
  id uuid references auth.users on delete cascade primary key,
  username text unique not null,
  email text,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  constraint username_length check (char_length(username) >= 3)
);

-- Habilitar Row Level Security (RLS) para perfiles
alter table public.profiles enable row level security;

-- Políticas de seguridad para perfiles
create policy "Cualquiera puede ver perfiles" on public.profiles for select using (true);
create policy "Usuarios pueden crear su propio perfil" on public.profiles for insert with check (auth.uid() = id);
create policy "Usuarios pueden editar su propio perfil" on public.profiles for update using (auth.uid() = id);

-- ==============================================================================
-- TABLA 2: SISTEMA DE AMIGOS Y SOLICITUDES (public.friends)
-- Maneja las relaciones entre usuarios y los estados de invitación social
-- ==============================================================================
create table public.friends (
  id bigint generated by default as identity primary key,
  user_id uuid references public.profiles(id) on delete cascade not null,  -- Quien envía o un lado de la relación
  friend_id uuid references public.profiles(id) on delete cascade not null, -- Quien recibe o el otro lado
  status text not null default 'pending',                                  -- 'pending' o 'accepted'
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  constraint no_auto_amistad check (user_id <> friend_id),                 -- Evita agregarse a uno mismo
  unique (user_id, friend_id)                                              -- Evita duplicar la misma solicitud
);

-- Habilitar RLS para amigos
alter table public.friends enable row level security;

-- Políticas de seguridad para amigos (Solo los involucrados pueden interactuar con sus registros)
create policy "Usuarios pueden gestionar sus propios amigos" 
on public.friends for all 
using (auth.uid() = user_id or auth.uid() = friend_id)
with check (auth.uid() = user_id or auth.uid() = friend_id);

-- ==============================================================================
-- TABLA 3: SALAS DE INTERCOMUNICACIÓN (public.rooms)
-- Registra las salas activas, su privacidad, contraseñas y el estado del canal de voz
-- ==============================================================================
create table public.rooms (
  id uuid default gen_random_uuid() primary key,
  room_code text unique not null,                                         -- Nombre o código visible (ej: ROVEX)
  host_id uuid references public.profiles(id) on delete cascade not null,  -- Creador (Piloto)
  guest_id uuid references public.profiles(id) on delete cascade,         -- Invitado (Copiloto)
  room_type text not null default 'public',                               -- 'public', 'password' o 'private'
  room_password text,                                                     -- Contraseña alfanumérica/caracteres
  sdp_offer jsonb,                                                        -- Oferta WebRTC (Señalización)
  sdp_answer jsonb,                                                       -- Respuesta WebRTC (Señalización)
  created_at timestamp with time zone default timezone('utc'::text, now()) not null,
  constraint tipo_sala_valido check (room_type in ('public', 'password', 'private'))
);

-- Habilitar RLS para salas
alter table public.rooms enable row level security;

-- Políticas de seguridad para desarrollo libre de salas
create policy "Acceso total a salas" on public.rooms for all using (true) with check (true);

-- ==============================================================================
-- TABLA 4: CANDIDATOS DE RED WEBRTC (public.ice_candidates)
-- Intercambio de IPs y puertos en tiempo real para enlazar los dispositivos
-- ==============================================================================
create table public.ice_candidates (
  id bigint generated by default as identity primary key,
  room_id uuid references public.rooms(id) on delete cascade not null,
  sender_id uuid references public.profiles(id) on delete cascade not null,
  candidate jsonb not null,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Habilitar RLS para candidatos ICE
alter table public.ice_candidates enable row level security;

-- Políticas de seguridad para desarrollo libre de candidatos ICE
create policy "Acceso total a candidatos ICE" on public.ice_candidates for all using (true) with check (true);
